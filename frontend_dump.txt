"use client";

import { useCallback, useEffect, useMemo, useState } from "react";
import toast from "react-hot-toast";
import { Search } from "lucide-react";
import { getAlerts, getErrorMessage, type Alert } from "@/lib/api";
import { timestamp } from "@/lib/format";
import { inputCls } from "@/lib/styles";
import { useShell } from "@/components/AppShell";
import { useLive } from "@/components/LiveProvider";
import AlertItem, { AlertItemSkeleton } from "@/components/AlertItem";
import ErrorState from "@/components/ErrorState";

type Filter = "all" | "change" | "status";
const FILTERS: Array<{ id: Filter; label: string }> = [
  { id: "all", label: "All" },
  { id: "change", label: "Changes" },
  { id: "status", label: "Problems" },
];

export default function AlertsPage() {
  const { refreshKey } = useShell();
  const { tick } = useLive();
  const [alerts, setAlerts] = useState<Alert[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [filter, setFilter] = useState<Filter>("all");
  const [q, setQ] = useState("");

  const load = useCallback(async () => {
    try {
      setAlerts(await getAlerts());
      setError(null);
    } catch (err) {
      const message = getErrorMessage(err);
      setError(message);
      toast.error(message, { id: "alerts-load-error" });
    }
  }, []);

  useEffect(() => { load(); }, [load, refreshKey, tick]);

  const visible = useMemo(() => {
    const needle = q.trim().toLowerCase();
    return [...(alerts ?? [])]
      .sort((a, b) => timestamp(b.sentAt) - timestamp(a.sentAt))
      .filter((a) => filter === "all" || (a.kind ?? "change") === filter)
      .filter((a) => !needle || `${a.monitor?.name ?? ""} ${a.summary} ${a.monitor?.url ?? ""}`.toLowerCase().includes(needle));
  }, [alerts, filter, q]);

  if (error && alerts === null) return <ErrorState message={error} onRetry={() => { setError(null); load(); }} />;

  return (
    <div className="animate-fade-in">
      <div className="flex flex-wrap items-center justify-between gap-2 border-b border-zinc-800 px-4 py-2">
        <div className="flex items-center gap-1">
          {FILTERS.map((f) => (
            <button
              key={f.id}
              type="button"
              onClick={() => setFilter(f.id)}
              aria-pressed={filter === f.id}
              className={`h-7 rounded-md px-2.5 text-[13px] transition-colors ${filter === f.id ? "bg-zinc-800 text-zinc-100" : "text-zinc-400 hover:bg-zinc-800/50 hover:text-zinc-200"}`}
            >
              {f.label}
            </button>
          ))}
        </div>
        <div className="relative w-full sm:w-64">
          <Search className="pointer-events-none absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-zinc-600" />
          <input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Search alerts" aria-label="Search alerts" className={`${inputCls} h-7 pl-8`} />
        </div>
      </div>

      {alerts === null ? (
        <ul>{Array.from({ length: 8 }).map((_, i) => <AlertItemSkeleton key={i} />)}</ul>
      ) : visible.length === 0 ? (
        <div className="px-6 py-20 text-center">
          <p className="text-[13px] font-medium text-zinc-200">{alerts.length === 0 ? "No alerts yet" : "Nothing matches"}</p>
          <p className="mt-1 text-[13px] text-zinc-500">
            {alerts.length === 0 ? "When a monitor detects a change, it will land here." : "Try a different filter or search term."}
          </p>
        </div>
      ) : (
        <ul>{visible.map((a) => <AlertItem key={a.id} alert={a} showUrl />)}</ul>
      )}
    </div>
  );
}

function GroupHeader({ icon, label, count }: { icon: StatusKind; label: string; count: number }) {
  return (
    <div className="flex h-9 items-center gap-2 border-b border-zinc-800/60 bg-zinc-900/40 px-4">

export default function MonitorsPage() {
  const router = useRouter();
  const { refreshKey, openAddMonitor } = useShell();
  const { tick } = useLive();
  const searchRef = useRef<HTMLInputElement>(null);
  const [monitors, setMonitors] = useState<Monitor[] | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [filter, setFilter] = useState<Filter>("all");
  const [q, setQ] = useState("");
  const [cursor, setCursor] = useState(-1);

  const load = useCallback(async () => {
    try {
      setMonitors(await getMonitors());
      setError(null);
    } catch (err) {
      const message = getErrorMessage(err);
      setError(message);
      toast.error(message, { id: "monitors-load-error" });
    }
  }, []);

  useEffect(() => { load(); }, [load, refreshKey, tick]);

  const { needsAttention, healthy } = useMemo(() => {
    const needle = q.trim().toLowerCase();
    const byRecent = (a: Monitor, b: Monitor) => timestamp(b.lastChangedAt) - timestamp(a.lastChangedAt) || a.name.localeCompare(b.name);
    const matches = (monitors ?? []).filter((m) => !needle || `${m.name} ${m.url}`.toLowerCase().includes(needle));
    const ok = (m: Monitor) => (m.status ?? "").toUpperCase() === "OK";
    return {
      needsAttention: filter === "ok" ? [] : matches.filter((m) => !ok(m)).sort(byRecent),
      healthy: filter === "problem" ? [] : matches.filter(ok).sort(byRecent),
    };
  }, [monitors, filter, q]);

  const visible = useMemo(() => [...needsAttention, ...healthy], [needsAttention, healthy]);

  // Linear-style keyboard navigation: j/k or arrows to move, Enter to open, / to search
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      const t = e.target as HTMLElement;
      if (t.isContentEditable || ["INPUT", "TEXTAREA", "SELECT"].includes(t.tagName) || e.metaKey || e.ctrlKey || e.altKey) return;
      const k = e.key.toLowerCase();
      if (k === "/") { e.preventDefault(); searchRef.current?.focus(); return; }
      if (!visible.length) return;
      if (k === "j" || k === "arrowdown") {
        e.preventDefault();
        setCursor((c) => { const n = Math.min(c + 1, visible.length - 1); document.getElementById(`row-${visible[n].id}`)?.scrollIntoView({ block: "nearest" }); return n; });
      } else if (k === "k" || k === "arrowup") {
        e.preventDefault();
        setCursor((c) => { const n = Math.max(c - 1, 0); document.getElementById(`row-${visible[n].id}`)?.scrollIntoView({ block: "nearest" }); return n; });
      } else if (k === "enter" && cursor >= 0 && visible[cursor]) {
        router.push(`/monitors/${visible[cursor].id}`);
      }
    };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [visible, cursor, router]);

  useEffect(() => { setCursor(-1); }, [filter, q]);

  if (error && monitors === null) return <ErrorState message={error} onRetry={() => { setError(null); load(); }} />;

  if (monitors === null) {
    return <div className="animate-fade-in">{Array.from({ length: 7 }).map((_, i) => <MonitorRowSkeleton key={i} />)}</div>;
  }

  if (monitors.length === 0) {
    return (
      <div className="flex h-full min-h-[60vh] flex-col items-center justify-center px-6 text-center">
        <div className="flex h-10 w-10 items-center justify-center rounded-lg border border-zinc-800 bg-zinc-900 text-zinc-500"><Globe className="h-5 w-5" /></div>
        <h2 className="mt-4 text-[15px] font-semibold tracking-tight">No monitors yet</h2>
        <p className="mt-1 max-w-sm text-[13px] text-zinc-500">Add a page to watch. You&rsquo;ll get a Telegram or email alert, with a visual diff, when it changes.</p>
        <button type="button" onClick={openAddMonitor} className={`${btnPrimary} mt-5`}>
          <Plus className="h-3.5 w-3.5" /> New monitor <span className="kbd !border-white/20 !bg-white/10 !text-white/80">C</span>
        </button>
      </div>
    );
  }

  const count = (f: Filter) => (f === "all" ? monitors.length : monitors.filter((m) => ((m.status ?? "").toUpperCase() === "OK") === (f === "ok")).length);
  const chip = (f: Filter, label: string) => (
    <button
      key={f}
      type="button"
      onClick={() => setFilter(f)}
      aria-pressed={filter === f}
      className={`inline-flex h-7 items-center gap-1.5 rounded-md px-2.5 text-[13px] transition-colors ${filter === f ? "bg-zinc-800 text-zinc-100" : "text-zinc-400 hover:bg-zinc-800/50 hover:text-zinc-200"}`}
    >
      {label} <span className="text-xs tabular-nums text-zinc-500">{count(f)}</span>
    </button>
  );

  return (
    <div className="animate-fade-in">
      <div className="flex flex-wrap items-center justify-between gap-2 border-b border-zinc-800 px-4 py-2">
        <div className="flex items-center gap-1">
          {chip("all", "All")}
          {chip("ok", "Healthy")}
          {chip("problem", "Problems")}
        </div>
        <div className="relative w-full sm:w-64">
          <Search className="pointer-events-none absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-zinc-600" />
          <input ref={searchRef} value={q} onChange={(e) => setQ(e.target.value)} placeholder="Filter…  ( / )" aria-label="Filter monitors" className={`${inputCls} h-7 pl-8`} />
        </div>
      </div>

      {visible.length === 0 ? (
        <p className="px-6 py-16 text-center text-[13px] text-zinc-500">No monitors match your filters.</p>
      ) : (
        <>
          {needsAttention.length > 0 && (
            <section>
              <GroupHeader icon="ERROR" label="Needs attention" count={needsAttention.length} />
              {needsAttention.map((m) => <MonitorRow key={m.id} monitor={m} onChanged={load} active={visible[cursor]?.id === m.id} />)}
            </section>
          )}
          {healthy.length > 0 && (
            <section>
              <GroupHeader icon="OK" label="Healthy" count={healthy.length} />
              {healthy.map((m) => <MonitorRow key={m.id} monitor={m} onChanged={load} active={visible[cursor]?.id === m.id} />)}
            </section>
          )}
          <p className="px-4 py-4 text-xs text-zinc-600">
            <span className="kbd">J</span> <span className="kbd">K</span> to move · <span className="kbd">↵</span> to open · <span className="kbd">/</span> to filter
          </p>
        </>
      )}
    </div>
  );
}

function MonitorDetail() {
  const { id } = useParams<{ id: string }>();
  const search = useSearchParams();
  const { scanning, tick, lastChange } = useLive();

  const [monitor, setMonitor] = useState<Monitor | null>(null);
  const [history, setHistory] = useState<Snapshot[] | null>(null);
  const [error, setError] = useState<string | null>(null);

  const [tab, setTab] = useState<Tab>("snapshot");
  const [selectedId, setSelectedId] = useState<string | null>(null);
  const [pair, setPair] = useState<Pair | null>(null);
  const [picked, setPicked] = useState<string[]>([]);
  const [queued, setQueued] = useState(false);

  const initialised = useRef(false);
  const handledChange = useRef(0);
  const mountedAt = useRef(Date.now());

  const load = useCallback(async () => {
    try {
      const [m, h] = await Promise.all([getMonitor(id), getHistory(id)]);
      setMonitor(m);
      setHistory(h);
      setError(null);
      return h;
    } catch (err) {
      setError(getErrorMessage(err));
      return null;
    }
  }, [id]);

  // Newest-first index, used to order a pair so v1 is always the older snapshot.
  const indexOf = useMemo(() => new Map((history ?? []).map((s, i): [string, number] => [s.id, i])), [history]);
  const orderPair = useCallback(
    (a: string, b: string): Pair => ((indexOf.get(a) ?? 0) > (indexOf.get(b) ?? 0) ? { v1: a, v2: b } : { v1: b, v2: a }),
    [indexOf]
  );

  // Initial load, and a refetch whenever the live stream reports a finished check
  useEffect(() => { load(); }, [load, tick]);

  // First time the history arrives: honour ?compare=a:b (from alert links), else show the latest change
  useEffect(() => {
    if (!history || initialised.current) return;
    initialised.current = true;
    setSelectedId(history[0]?.id ?? null);

    const [a, b] = (search.get("compare") ?? "").split(":");
    if (a && b && indexOf.has(a) && indexOf.has(b)) {
      setPair(orderPair(a, b));
      setTab("compare");
    } else if (history.length >= 2) {
      setPair({ v1: history[1].id, v2: history[0].id });
      setTab("compare");
    }
  }, [history, indexOf, orderPair, search]);

  // A change detected live for THIS monitor: jump straight to its visual diff
  useEffect(() => {
    // ignore changes that happened before this page opened (those arrive via ?compare=)
    if (!lastChange || lastChange.monitorId !== id || lastChange.at < mountedAt.current || lastChange.at === handledChange.current) return;
    handledChange.current = lastChange.at;
    load().then(() => {
      setPair({ v1: lastChange.prevSnapshotId, v2: lastChange.snapshotId });
      setSelectedId(lastChange.snapshotId);
      setTab("compare");
    });
  }, [lastChange, id, load]);

  // "Checking…" ends when the live stream reports a finished check (or after 45s)
  const isScanning = scanning.has(id);
  const tickAtQueue = useRef(0);
  useEffect(() => { if (queued && tick !== tickAtQueue.current && !isScanning) setQueued(false); }, [tick, queued, isScanning]);
  useEffect(() => {
    if (!queued) return;
    const t = setTimeout(() => setQueued(false), 45_000);
    return () => clearTimeout(t);
  }, [queued]);

  async function handleCheck() {
    try {
      await checkMonitor(id);
      tickAtQueue.current = tick;
      setQueued(true);
      toast.success("Check queued. Results appear here live.");
    } catch (err) {
      toast.error(getErrorMessage(err));
    }
  }

  function comparePrev(snapId: string) {
    const idx = indexOf.get(snapId);
    const older = idx === undefined ? undefined : history?.[idx + 1];
    if (!older) return;
    setPair({ v1: older.id, v2: snapId });
    setTab("compare");
  }

  function togglePick(snapId: string) {
    setPicked((p) => (p.includes(snapId) ? p.filter((x) => x !== snapId) : p.length >= 2 ? [p[1], snapId] : [...p, snapId]));
  }

  function compareSelected() {
    if (picked.length !== 2) return;
    setPair(orderPair(picked[0], picked[1]));
    setPicked([]);
    setTab("compare");
  }

  if (error && !monitor) return <ErrorState message={error} onRetry={() => { setError(null); load(); }} />;
  if (!monitor || !history) return <PageSkeleton />;

  const selected = history.find((s) => s.id === selectedId) ?? history[0];
  const adjacent = pair ? (indexOf.get(pair.v1) ?? -2) === (indexOf.get(pair.v2) ?? -9) + 1 : false;
  const checking = queued || isScanning;
  const latest = history[0];

  return (
    <div className="animate-fade-in lg:flex">
      {/* ───────── main column ───────── */}
      <div className="min-w-0 flex-1 px-6 py-5">
        <Link href="/monitors" className="inline-flex items-center gap-0.5 text-xs text-zinc-500 transition-colors hover:text-zinc-200">

"use client";

import { useMemo } from "react";
import { differenceInCalendarDays, format, startOfDay, subDays } from "date-fns";
import { timestamp } from "@/lib/format";

/** Tiny dependency-free bar chart: how many events happened on each of the last N days. */
export default function ActivityChart({
  dates,
  days = 14,
  label,
}: {
  dates: Array<string | null | undefined>;
  days?: number;
  label: string;
}) {
  const buckets = useMemo(() => {
    const out = Array.from({ length: days }, (_, i) => ({ day: startOfDay(subDays(new Date(), days - 1 - i)), count: 0 }));
    for (const raw of dates) {
      const t = timestamp(raw);
      if (!t) continue;
      const idx = differenceInCalendarDays(new Date(t), out[0].day);
      if (idx >= 0 && idx < days) out[idx].count++;
    }
    return out;
  }, [dates, days]);

  const max = Math.max(1, ...buckets.map((b) => b.count));
  const total = buckets.reduce((s, b) => s + b.count, 0);

  return (
    <div>
      <div className="flex items-baseline justify-between">
        <p className="text-[13px] font-medium text-zinc-200">{label}</p>
        <p className="text-xs tabular-nums text-zinc-500">{total} in {days}d</p>
      </div>
      <div className="mt-3 flex h-16 items-end gap-1" role="img" aria-label={`${label}: ${total} in the last ${days} days`}>
        {buckets.map((b) => (
          <div key={b.day.toISOString()} className="group relative flex h-full flex-1 items-end">
            <div
              className={`w-full rounded-[2px] transition-colors ${b.count ? "bg-indigo-500 group-hover:bg-indigo-400" : "bg-zinc-800/80"}`}
              style={{ height: `${b.count ? Math.max(14, (b.count / max) * 100) : 6}%` }}
            />
            <span className="pointer-events-none absolute -top-6 left-1/2 z-10 hidden -translate-x-1/2 whitespace-nowrap rounded border border-zinc-800 bg-zinc-900 px-1.5 py-0.5 text-[10px] text-zinc-200 group-hover:block">
              {format(b.day, "MMM d")} · {b.count}
            </span>
          </div>
        ))}
      </div>
      <div className="mt-1.5 flex justify-between text-[10px] text-zinc-600">
        <span>{format(buckets[0].day, "MMM d")}</span>
        <span>Today</span>
      </div>
    </div>
  );
}

function normalizeUrl(raw: string): string | null {
  const value = raw.trim();
  if (!value) return null;
  const withScheme = /^https?:\/\//i.test(value) ? value : `https://${value}`;
  try {
    const u = new URL(withScheme);
    if (!u.hostname.includes(".") && u.hostname !== "localhost") return null;
    return u.toString();
  } catch {
    return null;
  }
}

"use client";

import { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState, type ReactNode } from "react";
import { usePathname, useRouter } from "next/navigation";
import { ChevronRight, Menu, Plus } from "lucide-react";
import { btnPrimary } from "@/lib/styles";
import AddMonitorModal from "./AddMonitorModal";
import CommandPalette from "./CommandPalette";
import Sidebar from "./Sidebar";

interface ShellContextValue {
  /** Opens the "Create monitor" modal. */
  openAddMonitor: () => void;
  /** Increments whenever a monitor is created. Pages refetch when it changes. */
  refreshKey: number;
}

const ShellContext = createContext<ShellContextValue>({ openAddMonitor: () => {}, refreshKey: 0 });
export const useShell = () => useContext(ShellContext);

const TITLES: Record<string, string> = {
  "/": "Overview",
  "/monitors": "Monitors",
  "/alerts": "Alerts",
  "/settings": "Channels",
};

const titleFor = (pathname: string) =>
  TITLES[pathname] ?? (pathname.startsWith("/monitors/") ? "Monitors" : "Overview");

/** True when the keystroke is going into a form control, so single-key shortcuts must not fire. */
const isTyping = (t: EventTarget | null) => {
  const el = t as HTMLElement | null;
  return !!el && (el.isContentEditable || ["INPUT", "TEXTAREA", "SELECT"].includes(el.tagName));
};

export default function AppShell({ children }: { children: ReactNode }) {
  const pathname = usePathname();
  const router = useRouter();
  const [navOpen, setNavOpen] = useState(false);
  const [modalOpen, setModalOpen] = useState(false);
  const [paletteOpen, setPaletteOpen] = useState(false);
  const [refreshKey, setRefreshKey] = useState(0);
  const gPressedAt = useRef(0);

  const openAddMonitor = useCallback(() => setModalOpen(true), []);
  const closeModal = useCallback(() => setModalOpen(false), []);
  const closePalette = useCallback(() => setPaletteOpen(false), []);
  const openPalette = useCallback(() => setPaletteOpen(true), []);
  const handleCreated = useCallback(() => setRefreshKey((k) => k + 1), []);

  const ctx = useMemo(() => ({ openAddMonitor, refreshKey }), [openAddMonitor, refreshKey]);

  useEffect(() => { setNavOpen(false); }, [pathname]);

  useEffect(() => {
    if (!navOpen) return;
    const onKey = (e: KeyboardEvent) => { if (e.key === "Escape") setNavOpen(false); };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [navOpen]);

  // Keyboard-first, like Linear:  ⌘K palette · C create · G then O/M/A/C to navigate
  useEffect(() => {
    const onKey = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key.toLowerCase() === "k") {
        e.preventDefault();
        setPaletteOpen((o) => !o);
        return;
      }
      if (e.metaKey || e.ctrlKey || e.altKey || isTyping(e.target) || modalOpen || paletteOpen) return;

      const key = e.key.toLowerCase();
      if (key === "c") { e.preventDefault(); setModalOpen(true); return; }
      if (key === "g") { gPressedAt.current = Date.now(); return; }
      if (Date.now() - gPressedAt.current < 1200) {
        const dest: Record<string, string> = { o: "/", m: "/monitors", a: "/alerts", c: "/settings" };
        if (dest[key]) { e.preventDefault(); gPressedAt.current = 0; router.push(dest[key]); }
      }
    };
    document.addEventListener("keydown", onKey);
    return () => document.removeEventListener("keydown", onKey);
  }, [router, modalOpen, paletteOpen]);

  const sidebar = (onNavigate?: () => void) => (
    <Sidebar onNavigate={onNavigate} onSearch={openPalette} onCreate={openAddMonitor} refreshKey={refreshKey} />
  );

  return (
    <ShellContext.Provider value={ctx}>
      <div className="flex h-dvh bg-canvas text-zinc-100">
        <aside className="hidden w-60 shrink-0 lg:block">{sidebar()}</aside>

        {navOpen && (
          <div className="fixed inset-0 z-40 lg:hidden" role="dialog" aria-modal="true" aria-label="Navigation">
            <div className="absolute inset-0 animate-fade-in bg-black/70" onClick={() => setNavOpen(false)} />
            <aside className="absolute inset-y-0 left-0 w-64 border-r border-zinc-800 bg-canvas">{sidebar(() => setNavOpen(false))}</aside>
          </div>
        )}

        {/* Content lives in an inset, bordered panel, the signature Linear layout */}
        <div className="flex min-w-0 flex-1 flex-col lg:py-2 lg:pr-2">
          <div className="flex min-h-0 flex-1 flex-col overflow-hidden bg-zinc-950 lg:rounded-lg lg:border lg:border-zinc-800">
            <header className="flex h-11 shrink-0 items-center justify-between border-b border-zinc-800 px-4">
              <div className="flex min-w-0 items-center gap-2 text-[13px]">
                <button
                  type="button"
                  onClick={() => setNavOpen(true)}
                  aria-label="Open navigation"
                  className="-ml-1.5 flex h-7 w-7 items-center justify-center rounded-md text-zinc-400 hover:bg-zinc-800/70 hover:text-zinc-100 lg:hidden"
                >
                  <Menu className="h-4 w-4" />
                </button>
                <span className="hidden text-zinc-500 sm:inline">Aegis Watch</span>
                <ChevronRight className="hidden h-3.5 w-3.5 text-zinc-700 sm:inline" />
                <h1 className="truncate font-medium text-zinc-100">{titleFor(pathname)}</h1>
              </div>
              <button type="button" onClick={openAddMonitor} aria-label="Create monitor" className={btnPrimary}>
                <Plus className="h-3.5 w-3.5" />

              </button>
            </header>

            <main className="scroll-thin min-h-0 flex-1 overflow-y-auto">{children}</main>
          </div>
        </div>

        <AddMonitorModal open={modalOpen} onClose={closeModal} onCreated={handleCreated} />

"use client";

import { useEffect, useRef, useState, type FormEvent } from "react";
import { Loader2, Send, Sparkles } from "lucide-react";
import { askMonitor, getErrorMessage } from "@/lib/api";

const SUGGESTIONS = [
  "What changed most recently?",
  "Did any prices or numbers change?",
  "Summarize everything that changed so far",
];

interface Turn { q: string; a?: string; error?: string }

export default function AskAI({ monitorId, disabled }: { monitorId: string; disabled?: boolean }) {
  const [turns, setTurns] = useState<Turn[]>([]);
  const [input, setInput] = useState("");
  const [busy, setBusy] = useState(false);
  const endRef = useRef<HTMLDivElement>(null);

  useEffect(() => { endRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" }); }, [turns]);

  async function ask(question: string) {
    const q = question.trim();
    if (q.length < 3 || busy) return;
    setBusy(true);
    setInput("");
    setTurns((t) => [...t, { q }]);
    try {
      const { answer } = await askMonitor(monitorId, q);
      setTurns((t) => t.map((x, i) => (i === t.length - 1 ? { ...x, a: answer } : x)));
    } catch (err) {
      setTurns((t) => t.map((x, i) => (i === t.length - 1 ? { ...x, error: getErrorMessage(err) } : x)));
    } finally {
      setBusy(false);
    }
  }

  function onSubmit(e: FormEvent) {
    e.preventDefault();
    ask(input);
  }

  return (
    <div className="overflow-hidden rounded-lg border border-zinc-800 bg-zinc-900/30">
      <div className="flex items-center gap-2 border-b border-zinc-800 px-4 py-3">
        <Sparkles className="h-4 w-4 text-indigo-300" />
        <h3 className="text-sm font-medium text-zinc-100">Ask this page&rsquo;s history</h3>
        <span className="ml-auto text-xs text-zinc-600">Gemini · last 6 snapshots</span>
      </div>

      <div className="scroll-thin max-h-[420px] min-h-[220px] space-y-4 overflow-y-auto px-4 py-4">
        {turns.length === 0 && (
          <div className="py-4 text-center">
            <p className="text-sm text-zinc-400">Ask anything about how this page has changed over time.</p>
            <div className="mt-4 flex flex-wrap justify-center gap-2">
              {SUGGESTIONS.map((s) => (
                <button
                  key={s}
                  type="button"
                  disabled={disabled || busy}
                  onClick={() => ask(s)}
                  className="rounded-full border border-zinc-800 bg-zinc-900 px-3 py-1.5 text-xs text-zinc-300 transition-colors hover:border-indigo-500/40 hover:text-indigo-200 disabled:opacity-50"
                >
                  {s}
                </button>
              ))}
            </div>
          </div>
        )}

        {turns.map((t, i) => (
          <div key={i} className="space-y-2">
            <div className="ml-auto w-fit max-w-[85%] rounded-2xl rounded-br-sm bg-indigo-500/20 px-3.5 py-2 text-sm text-indigo-100">{t.q}</div>
            <div className="max-w-[92%] rounded-2xl rounded-bl-sm border border-zinc-800 bg-zinc-900 px-3.5 py-2.5 text-sm leading-relaxed text-zinc-200">
              {t.error ? (
                <span className="text-red-300">{t.error}</span>
              ) : t.a ? (
                t.a
              ) : (
                <span className="inline-flex items-center gap-2 text-zinc-500"><Loader2 className="h-3.5 w-3.5 animate-spin" /> Reading the snapshots…</span>
              )}
            </div>
          </div>
        ))}
        <div ref={endRef} />
      </div>

      <form onSubmit={onSubmit} className="flex items-center gap-2 border-t border-zinc-800 p-3">
        <input
          value={input}
          onChange={(e) => setInput(e.target.value)}
          maxLength={300}

import { useEffect, useMemo, useRef, useState, type KeyboardEvent } from "react";

import { Activity, Bell, CornerDownLeft, Globe, LayoutDashboard, Plus, Search, Send, type LucideIcon } from "lucide-react";
import { getMonitors, type Monitor } from "@/lib/api";

interface Item { id: string; label: string; hint?: string; icon: LucideIcon; run: () => void }

export default function CommandPalette({
  open,
  onClose,
  onCreate,
}: {
  open: boolean;
  onClose: () => void;
  onCreate: () => void;
}) {
  const router = useRouter();
  const inputRef = useRef<HTMLInputElement>(null);
  const [q, setQ] = useState("");
  const [active, setActive] = useState(0);
  const [monitors, setMonitors] = useState<Monitor[]>([]);

  useEffect(() => {
    if (!open) return;
    setQ("");
    setActive(0);
    inputRef.current?.focus();
    getMonitors().then(setMonitors).catch(() => {});
  }, [open]);

  const items = useMemo<Item[]>(() => {
    const go = (href: string) => () => { router.push(href); onClose(); };
    const base: Item[] = [
      { id: "new", label: "Create monitor", hint: "Action", icon: Plus, run: () => { onClose(); onCreate(); } },
      { id: "dash", label: "Dashboard", hint: "Go to", icon: LayoutDashboard, run: go("/") },
      { id: "mons", label: "Monitors", hint: "Go to", icon: Activity, run: go("/monitors") },
      { id: "alerts", label: "Alerts", hint: "Go to", icon: Bell, run: go("/alerts") },
      { id: "channels", label: "Notification channels", hint: "Go to", icon: Send, run: go("/settings") },
    ];
    const mons: Item[] = monitors.map((m) => ({
      id: `m-${m.id}`,
      label: m.name,
      hint: m.url.replace(/^https?:\/\//, ""),
      icon: Globe,
      run: go(`/monitors/${m.id}`),
    }));
    const needle = q.trim().toLowerCase();
    const all = [...base, ...mons];
    return needle ? all.filter((i) => `${i.label} ${i.hint ?? ""}`.toLowerCase().includes(needle)) : all;
  }, [monitors, q, router, onClose, onCreate]);

  if (!open) return null;

  function onKeyDown(e: KeyboardEvent<HTMLInputElement>) {
    if (e.key === "ArrowDown") { e.preventDefault(); setActive((a) => Math.min(a + 1, items.length - 1)); }
    else if (e.key === "ArrowUp") { e.preventDefault(); setActive((a) => Math.max(a - 1, 0)); }
    else if (e.key === "Enter") { e.preventDefault(); items[active]?.run(); }
    else if (e.key === "Escape") onClose();
  }

  return (
    <div
      className="fixed inset-0 z-[60] flex animate-fade-in items-start justify-center bg-black/70 px-4 pt-[15vh] backdrop-blur-sm"
      onMouseDown={(e) => { if (e.target === e.currentTarget) onClose(); }}

        className="w-full max-w-lg animate-modal-in overflow-hidden rounded-xl border border-zinc-800 bg-zinc-950 shadow-2xl shadow-black/60">
        <div className="flex items-center gap-3 border-b border-zinc-800 px-4">
          <Search className="h-4 w-4 text-zinc-500" />
          <input
            ref={inputRef}
            value={q}
            onChange={(e) => { setQ(e.target.value); setActive(0); }}
            onKeyDown={onKeyDown}
            placeholder="Search monitors, jump to a page…"
            className="h-11 w-full bg-transparent text-sm text-zinc-100 placeholder:text-zinc-600 focus:outline-none"
          />
          <kbd className="rounded border border-zinc-700 px-1.5 py-0.5 text-[10px] text-zinc-500">ESC</kbd>
        </div>
        <ul className="scroll-thin max-h-80 overflow-y-auto p-2">
          {items.length === 0 && <li className="px-3 py-8 text-center text-sm text-zinc-500">No results</li>}
          {items.map((item, i) => {
            const Icon = item.icon;
            return (
              <li key={item.id}>
                <button
                  type="button"
                  onMouseEnter={() => setActive(i)}
                  onClick={item.run}
                  className={`flex w-full items-center gap-3 rounded-lg px-3 py-2.5 text-left text-sm transition-colors ${i === active ? "bg-indigo-500/10 text-indigo-200" : "text-zinc-300"}`}
                >
                  <Icon className="h-4 w-4 shrink-0 text-zinc-500" />
                  <span className="truncate">{item.label}</span>
                  {item.hint && <span className="ml-auto truncate text-xs text-zinc-600">{item.hint}</span>}
                  {i === active && <CornerDownLeft className="h-3.5 w-3.5 shrink-0 text-indigo-300" />}
                </button>
              </li>
            );
          })}
        </ul>
      </div>
    </div>

/* ───────── text diff: parse the API's <ins>/<del>/<span> HTML into React nodes (no innerHTML) ───────── */

type Seg = { kind: "ins" | "del" | "same"; text: string };

function parseDiff(html: string): Seg[] {
  const out: Seg[] = [];
  const re = /<(ins|del|span)>([\s\S]*?)<\/\1>/g;
  let m: RegExpExecArray | null;
  while ((m = re.exec(html))) {
    out.push({ kind: m[1] === "span" ? "same" : (m[1] as "ins" | "del"), text: m[2].replace(/&lt;/g, "<") });
  }
  return out;
}

const CONTEXT = 110;

function TextDiff({ html }: { html: string }) {
  const [full, setFull] = useState(false);
  const segs = useMemo(() => parseDiff(html), [html]);
  const changes = segs.filter((s) => s.kind !== "same");

  if (changes.length === 0) {
    return <p className="px-4 py-10 text-center text-sm text-zinc-500">No textual differences between these two snapshots.</p>;
  }

"use client";

import { createContext, useContext, useEffect, useMemo, useState, type ReactNode } from "react";
import { useRouter } from "next/navigation";
import toast from "react-hot-toast";
import { ArrowRight, Sparkles } from "lucide-react";
import { API_BASE, API_KEY } from "@/lib/api";

export interface ChangeEvent {
  monitorId: string;
  name: string;
  url: string;
  summary: string;
  snapshotId: string;
  prevSnapshotId: string;
  changePercent: number;
  at: number;
}

interface LiveValue {
  connected: boolean;
  /** Monitor ids being checked right now. */
  scanning: ReadonlySet<string>;
  /** Increments after every finished check; pages refetch when it changes. */
  tick: number;
  lastChange: ChangeEvent | null;
}

const LiveContext = createContext<LiveValue>({ connected: false, scanning: new Set(), tick: 0, lastChange: null });
export const useLive = () => useContext(LiveContext);

/**
 * Real-time updates over Server-Sent Events. EventSource cannot send the x-api-key
 * header, so this reads the stream with fetch() and parses the frames itself.
 */
export function LiveProvider({ children }: { children: ReactNode }) {
  const router = useRouter();
  const [connected, setConnected] = useState(false);
  const [scanning, setScanning] = useState<ReadonlySet<string>>(new Set());
  const [tick, setTick] = useState(0);
  const [lastChange, setLastChange] = useState<ChangeEvent | null>(null);

  useEffect(() => {
    const ctrl = new AbortController();
    let retry: ReturnType<typeof setTimeout> | undefined;

    const setScan = (id: string, on: boolean) =>
      setScanning((prev) => {
        const next = new Set(prev);
        if (on) next.add(id); else next.delete(id);
        return next;
      });

    function handle(event: string, data: any) {
      if (event === "check:started") setScan(data.monitorId, true);
      if (event === "check:finished") { setScan(data.monitorId, false); setTick((t) => t + 1); }
      if (event === "change:detected") {
        setLastChange({ ...data, at: Date.now() });
        setTick((t) => t + 1);
        toast.custom(
          (t) => (
            <div className={`${t.visible ? "animate-slide-up" : "opacity-0"} pointer-events-auto w-80 rounded-lg border border-zinc-800 bg-zinc-900 p-4 shadow-2xl shadow-indigo-500/10`}>
              <div className="flex items-center gap-2 text-xs font-medium text-indigo-300">
                <Sparkles className="h-3.5 w-3.5" /> Change detected
              </div>
              <p className="mt-1.5 truncate text-sm font-medium text-zinc-100">{data.name}</p>
              <p className="mt-1 line-clamp-2 text-sm text-zinc-400">{data.summary}</p>
              <button
                type="button"
                onClick={() => { toast.dismiss(t.id); router.push(`/monitors/${data.monitorId}?compare=${data.prevSnapshotId}:${data.snapshotId}`); }}
                className="mt-3 inline-flex items-center gap-1.5 text-xs font-medium text-indigo-300 hover:text-indigo-200"
              >
                View visual diff <ArrowRight className="h-3.5 w-3.5" />
              </button>
            </div>
          ),
          { duration: 9000, id: `change-${data.snapshotId}` }
        );
      }
    }

    async function connect() {
      try {
        const res = await fetch(`${API_BASE}/events`, {
          headers: { "x-api-key": API_KEY, Accept: "text/event-stream" },
          signal: ctrl.signal,
          cache: "no-store",
        });
        if (!res.ok || !res.body) throw new Error(`stream failed (${res.status})`);
        setConnected(true);

        const reader = res.body.getReader();
        const decoder = new TextDecoder();
        let buf = "";
        for (;;) {
          const { done, value } = await reader.read();
          if (done) break;
          buf += decoder.decode(value, { stream: true });
          let i: number;
          while ((i = buf.indexOf("\n\n")) !== -1) {
            const frame = buf.slice(0, i);
            buf = buf.slice(i + 2);
            let event = "message";
            let payload = "";
            for (const line of frame.split("\n")) {
              if (line.startsWith("event:")) event = line.slice(6).trim();
              else if (line.startsWith("data:")) payload += line.slice(5).trim();
            }
            if (payload) {
              try { handle(event, JSON.parse(payload)); } catch { /* ignore malformed frame */ }
            }
          }
        }
      } catch {
        /* network drop or abort: fall through to reconnect */
      }
      setConnected(false);
      if (!ctrl.signal.aborted) retry = setTimeout(connect, 3000);
    }

    connect();
    return () => { ctrl.abort(); clearTimeout(retry); };
  }, [router]);

  const value = useMemo(() => ({ connected, scanning, tick, lastChange }), [connected, scanning, tick, lastChange]);
  return <LiveContext.Provider value={value}>{children}</LiveContext.Provider>;
}

export type StatusKind = "OK" | "ERROR" | "BLOCKED" | "PENDING" | "SCANNING";

export function normalizeStatus(status?: string | null): Exclude<StatusKind, "SCANNING"> {
  const n = (status ?? "").toUpperCase();
  return n === "OK" || n === "ERROR" || n === "BLOCKED" ? n : "PENDING";
}

};

// --- Generic Fetch ---
async function fetchApi<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
  const res = await fetch(`${API_URL}${endpoint}`, {
    ...options,
    headers: { ...headers, ...options.headers },
    cache: "no-store",
  });

  if (!res.ok) {
    let errorMsg = `Request failed with status ${res.status}`;
    try {
      const data = await res.json();
      if (data.error) errorMsg = data.error;
    } catch {}
    throw new Error(errorMsg);
  }
  
  if (res.status === 204) return {} as T;
  return res.json();
}

import { format, formatDistanceToNow } from "date-fns";

function toDate(value?: string | null): Date | null {
  if (!value) return null;
  const d = new Date(value);
  return Number.isNaN(d.getTime()) ? null : d;
}

export function timeAgo(value?: string | null, fallback = "Never"): string {
  const d = toDate(value);
  return d ? formatDistanceToNow(d, { addSuffix: true }) : fallback;
}

/** Compact relative time like Linear's list rows: now, 5m, 3h, 2d, Oct 4. */
export function shortAgo(value?: string | null, fallback = "—"): string {
  const d = toDate(value);
  if (!d) return fallback;
  const s = Math.max(0, Math.floor((Date.now() - d.getTime()) / 1000));
  if (s < 45) return "now";
  if (s < 3600) return `${Math.max(1, Math.round(s / 60))}m`;
  if (s < 86400) return `${Math.round(s / 3600)}h`;
  if (s < 7 * 86400) return `${Math.round(s / 86400)}d`;
  return format(d, "MMM d");
}

export function fullDate(value?: string | null): string {
  const d = toDate(value);
  return d ? format(d, "PPpp") : "";
}

export function timestamp(value?: string | null): number {
  return toDate(value)?.getTime() ?? 0;
}

export function formatInterval(mins: number): string {
  if (mins % 1440 === 0) {
    const days = mins / 1440;
    return days === 1 ? "daily" : `every ${days} days`;
  }
  if (mins % 60 === 0) {
    const hours = mins / 60;
    return hours === 1 ? "hourly" : `every ${hours} hours`;
  }
  return `every ${mins} min`;
}

/** 15 -> "15m", 60 -> "1h", 1440 -> "1d" */
export function shortInterval(mins: number): string {
  if (mins % 1440 === 0) return `${mins / 1440}d`;
  if (mins % 60 === 0) return `${mins / 60}h`;
  return `${mins}m`;
}
