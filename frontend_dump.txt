type Tone = "default" | "ok" | "error";

const ICON_TONE: Record<Tone, string> = {
  default: "text-zinc-400",
  ok: "text-emerald-400",
  error: "text-red-400",
};

function StatCard({
  label,
  value,
  icon: Icon,
  loading,
  tone = "default",
}: {
  label: string;
  value: number;
  icon: LucideIcon;
  loading: boolean;
  tone?: Tone;
}) {
  const valueTone = tone === "error" && value > 0 ? "text-red-300" : "text-zinc-100";
  return (
    <div className="rounded-xl border border-zinc-800/80 bg-gradient-to-b from-zinc-900/70 to-zinc-900/30 p-4 sm:p-5">
      <div className="flex items-center justify-between">
        <p className="text-sm text-zinc-400">{label}</p>
        <Icon className={`h-4 w-4 ${ICON_TONE[tone]}`} />
      </div>
      <div className="mt-3 h-9">
        {loading ? (
          <Skeleton className="h-8 w-16" />
        ) : (
          <p className={`text-3xl font-semibold tabular-nums tracking-tight ${valueTone}`}>
            {value}
          </p>
        )}
      </div>
    </div>
  );
}

function SectionHeader({
  title,
  href,
  linkLabel,
}: {
  title: string;
  href: string;
  linkLabel: string;
}) {
  return (
    <div className="mb-3 flex items-center justify-between">
      <h2 className="text-sm font-medium text-zinc-100">{title}</h2>
      <Link
        href={href}
        className="text-xs text-zinc-500 transition-colors hover:text-zinc-200 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-zinc-500"
      >
        {linkLabel}
      </Link>
    </div>
  );
}

export default function DashboardPage() {
  const { refreshKey, openAddMonitor } = useShell();
  const [monitors, setMonitors] = useState<Monitor[] | null>(null);
  const [alerts, setAlerts] = useState<Alert[] | null>(null);
  const [error, setError] = useState<string | null>(null);

  const load = useCallback(async () => {
    try {
      const [m, a] = await Promise.all([getMonitors(), getAlerts()]);
      setMonitors(m);
      setAlerts(a);
      setError(null);
    } catch (err) {
      const message = getErrorMessage(err);
      setError(message);
      toast.error(message, { id: "dashboard-load-error" });
    }
  }, []);

  useEffect(() => {
    load();
  }, [load, refreshKey]);

  const retry = () => {
    setError(null);
    load();
  };

  const loading = !error && (monitors === null || alerts === null);

  if (error && (monitors === null || alerts === null)) {
    return <ErrorState message={error} onRetry={retry} />;
  }

  const list = monitors ?? [];
  const alertList = alerts ?? [];
  const isStatus = (m: Monitor, s: string) => (m.status ?? "").toUpperCase() === s;

  const healthy = list.filter((m) => isStatus(m, "OK")).length;
  const errored = list.filter((m) => isStatus(m, "ERROR")).length;

  const recent = [...alertList]
    .sort((a, b) => timestamp(b.sentAt) - timestamp(a.sentAt))
    .slice(0, 6);

  // Surface problems first, then the most recently changed pages
  const topMonitors = [...list]
    .sort((a, b) => {
      const ae = isStatus(a, "ERROR") ? 1 : 0;
      const be = isStatus(b, "ERROR") ? 1 : 0;
      if (ae !== be) return be - ae;
      return timestamp(b.lastChangedAt) - timestamp(a.lastChangedAt);
    })
    .slice(0, 3);

  return (
    <div className="space-y-8">
      <section aria-label="Summary" className="grid grid-cols-2 gap-3 sm:gap-4 lg:grid-cols-4">
        <StatCard label="Total monitors" value={list.length} icon={Globe} loading={loading} />
        <StatCard label="Healthy" value={healthy} icon={CheckCircle2} loading={loading} tone="ok" />
        <StatCard label="Errors" value={errored} icon={AlertTriangle} loading={loading} tone="error" />
        <StatCard label="Total alerts" value={alertList.length} icon={Bell} loading={loading} />
      </section>

      <section aria-label="Recent activity">
        <SectionHeader title="Recent activity" href="/alerts" linkLabel="View all alerts" />
        <div className="overflow-hidden rounded-xl border border-zinc-800/80 bg-zinc-900/40">
          {loading ? (
            <ul className="divide-y divide-zinc-800/60">
              {Array.from({ length: 4 }).map((_, i) => (
                <AlertItemSkeleton key={i} />
              ))}
            </ul>
          ) : recent.length === 0 ? (
            <div className="px-6 py-10 text-center">
              <p className="text-sm font-medium text-zinc-200">No alerts yet</p>
              <p className="mt-1 text-sm text-zinc-500">
                Changes show up here as soon as a monitor detects one.
              </p>
            </div>
          ) : (
            <ul className="divide-y divide-zinc-800/60">
              {recent.map((alert) => (
                <AlertItem key={alert.id} alert={alert} />
              ))}
            </ul>
          )}
        </div>
      </section>

      <section aria-label="Monitors">
        <SectionHeader title="Monitors" href="/monitors" linkLabel="View all monitors" />
        {loading ? (
          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
            {Array.from({ length: 3 }).map((_, i) => (
              <MonitorCardSkeleton key={i} />
            ))}
          </div>
        ) : topMonitors.length === 0 ? (
          <div className="flex flex-col items-center rounded-xl border border-dashed border-zinc-800 px-6 py-10 text-center">
            <p className="text-sm font-medium text-zinc-200">No monitors yet</p>
            <p className="mt-1 text-sm text-zinc-500">
              Add a page to watch and we&rsquo;ll alert you when it changes.
            </p>
            <button type="button" onClick={openAddMonitor} className={`${btnSecondary} mt-4`}>
              <Plus className="h-3.5 w-3.5" />
              Create monitor
            </button>
          </div>
        ) : (
          <div className="grid gap-4 md:grid-cols-2 xl:grid-cols-3">
            {topMonitors.map((m) => (
              <MonitorCard key={m.id} monitor={m} onChanged={load} />
            ))}
          </div>
        )}
      </section>
    </div>
  );
}

"use client";

import { useEffect, useId, useRef, useState, type FormEvent } from "react";
import toast from "react-hot-toast";
import { ChevronDown, Loader2, X } from "lucide-react";
import { createMonitor, getErrorMessage } from "@/lib/api";
import { btnGhost, btnPrimary } from "@/lib/styles";

interface Props {
  open: boolean;
  onClose: () => void;
  onCreated?: () => void;
}

const INTERVALS = [
  { value: 15, label: "Every 15 minutes" },
  { value: 30, label: "Every 30 minutes" },
  { value: 60, label: "Every hour" },
  { value: 1440, label: "Every day" },
];

const inputBase =
  "h-10 w-full rounded-md border bg-zinc-900/60 px-3 text-sm text-zinc-100 placeholder:text-zinc-600 transition-colors focus:outline-none focus:ring-2 focus:ring-zinc-500/50 disabled:opacity-60";
const inputCls = (hasError: boolean) =>
  `${inputBase} ${hasError ? "border-red-500/60" : "border-zinc-800 focus:border-zinc-600"}`;

function normalizeUrl(raw: string): string | null {
  const value = raw.trim();
  if (!value) return null;
  const withScheme = /^https?:\/\//i.test(value) ? value : `https://${value}`;
  try {
    const u = new URL(withScheme);
    if (!u.hostname.includes(".") && u.hostname !== "localhost") return null;
    return u.toString();
  } catch {
    return null;
  }
}

export default function AddMonitorModal({ open, onClose, onCreated }: Props) {
  const uid = useId();
  const nameRef = useRef<HTMLInputElement>(null);

  const [name, setName] = useState("");
  const [url, setUrl] = useState("");
  const [intervalMins, setIntervalMins] = useState(60);
  const [sensitivity, setSensitivity] = useState(50);
  const [errors, setErrors] = useState<{ name?: string; url?: string }>({});
  const [submitting, setSubmitting] = useState(false);

  // Reset the form and focus the first field each time the modal opens
  useEffect(() => {
    if (!open) return;
    setName("");
    setUrl("");
    setIntervalMins(60);
    setSensitivity(50);
    setErrors({});
    setSubmitting(false);
    const t = setTimeout(() => nameRef.current?.focus(), 30);
    return () => clearTimeout(t);
  }, [open]);

  // Escape to close and lock page scroll while open
  useEffect(() => {
    if (!open) return;
    const onKey = (e: KeyboardEvent) => {
      if (e.key === "Escape" && !submitting) onClose();
    };
    document.addEventListener("keydown", onKey);
    const prevOverflow = document.body.style.overflow;
    document.body.style.overflow = "hidden";
    return () => {
      document.removeEventListener("keydown", onKey);
      document.body.style.overflow = prevOverflow;
    };
  }, [open, submitting, onClose]);

  if (!open) return null;

  async function handleSubmit(e: FormEvent<HTMLFormElement>) {
    e.preventDefault();
    if (submitting) return;

    const cleanName = name.trim();
    const normalizedUrl = normalizeUrl(url);
    const next: { name?: string; url?: string } = {};

    if (!cleanName) next.name = "Enter a name for this monitor.";
    if (!url.trim()) next.url = "Enter the URL of the page to watch.";
    else if (!normalizedUrl) next.url = "Enter a valid URL, like https://example.com/pricing.";

    setErrors(next);
    if (Object.keys(next).length > 0) return;

    setSubmitting(true);
    try {
      await createMonitor({
        name: cleanName,
        url: normalizedUrl as string,
        intervalMins,
        sensitivity,
      });
      toast.success("Monitor created");
      onCreated?.();
      onClose();
    } catch (err) {
      toast.error(getErrorMessage(err));
      setSubmitting(false);
    }
  }

  const nameId = `${uid}-name`;
  const urlId = `${uid}-url`;
  const intervalId = `${uid}-interval`;
  const sensitivityId = `${uid}-sensitivity`;
  const titleId = `${uid}-title`;

  return (
    <div
      className="fixed inset-0 z-50 flex animate-fade-in items-end justify-center bg-black/70 backdrop-blur-sm sm:items-center sm:p-4"
      onMouseDown={(e) => {
        if (e.target === e.currentTarget && !submitting) onClose();
      }}
    >
      <div
        role="dialog"
        aria-modal="true"
        aria-labelledby={titleId}
        className="max-h-[92vh] w-full max-w-md animate-modal-in overflow-y-auto rounded-t-2xl border border-zinc-800 bg-zinc-950 shadow-2xl shadow-black/50 sm:rounded-2xl"
      >
        <div className="flex items-start justify-between gap-4 border-b border-zinc-800/80 px-5 py-4">
          <div>
            <h2 id={titleId} className="text-base font-semibold tracking-tight">
              Create monitor
            </h2>
            <p className="mt-1 text-sm text-zinc-400">
              We&rsquo;ll check this page on a schedule and alert you when it changes.
            </p>
          </div>
          <button
            type="button"
            onClick={onClose}
            disabled={submitting}
            aria-label="Close"
            className="-mr-1.5 -mt-1 inline-flex h-8 w-8 shrink-0 items-center justify-center rounded-md text-zinc-500 transition-colors hover:bg-zinc-800/70 hover:text-zinc-100 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-zinc-500 disabled:opacity-50"
          >
            <X className="h-4 w-4" />
          </button>
        </div>

        <form onSubmit={handleSubmit} noValidate className="space-y-5 px-5 py-5">
          <div>
            <label htmlFor={nameId} className="mb-1.5 block text-sm font-medium text-zinc-200">
              Name
            </label>
            <input
              ref={nameRef}
              id={nameId}
              type="text"
              value={name}
              onChange={(e) => setName(e.target.value)}
              placeholder="Pricing page"
              disabled={submitting}
              autoComplete="off"
              aria-invalid={!!errors.name}
              aria-describedby={errors.name ? `${nameId}-error` : undefined}
              className={inputCls(!!errors.name)}
            />
            {errors.name && (
              <p id={`${nameId}-error`} className="mt-1.5 text-xs text-red-400">
                {errors.name}
              </p>
            )}
          </div>

          <div>
            <label htmlFor={urlId} className="mb-1.5 block text-sm font-medium text-zinc-200">
              URL
            </label>
            <input
              id={urlId}
              type="text"
              inputMode="url"
              value={url}
              onChange={(e) => setUrl(e.target.value)}
              placeholder="https://example.com/pricing"
              disabled={submitting}
              autoComplete="off"
              autoCapitalize="off"
              spellCheck={false}
              aria-invalid={!!errors.url}
              aria-describedby={errors.url ? `${urlId}-error` : undefined}
              className={inputCls(!!errors.url)}
            />
            {errors.url && (
              <p id={`${urlId}-error`} className="mt-1.5 text-xs text-red-400">
                {errors.url}
              </p>
            )}
          </div>

          <div>
            <label htmlFor={intervalId} className="mb-1.5 block text-sm font-medium text-zinc-200">
              Check interval
            </label>
            <div className="relative">
              <select
                id={intervalId}
                value={intervalMins}
                onChange={(e) => setIntervalMins(Number(e.target.value))}
                disabled={submitting}
                className={`${inputCls(false)} appearance-none pr-9`}
              >
                {INTERVALS.map((o) => (
                  <option key={o.value} value={o.value}>
                    {o.label}
                  </option>
                ))}
              </select>
              <ChevronDown className="pointer-events-none absolute right-3 top-1/2 h-4 w-4 -translate-y-1/2 text-zinc-500" />
            </div>
          </div>

          <div>
            <div className="mb-2.5 flex items-center justify-between">
              <label htmlFor={sensitivityId} className="text-sm font-medium text-zinc-200">
                Sensitivity
              </label>
              <span className="text-sm tabular-nums text-zinc-400">{sensitivity}%</span>
            </div>
            <input
              id={sensitivityId}
              type="range"
              min={1}
              max={100}
              step={1}
              value={sensitivity}
              onChange={(e) => setSensitivity(Number(e.target.value))}
              disabled={submitting}
              className="range w-full"
              style={{
                background: `linear-gradient(to right, #e4e4e7 ${sensitivity}%, #27272a ${sensitivity}%)`,
              }}
            />
            <p className="mt-2 text-xs text-zinc-500">
              Higher sensitivity flags smaller changes on the page.
            </p>
          </div>

          <div className="flex flex-col-reverse gap-2 pt-1 sm:flex-row sm:justify-end">
            <button type="button" onClick={onClose} disabled={submitting} className={btnGhost}>
              Cancel
            </button>
            <button type="submit" disabled={submitting} className={btnPrimary}>
              {submitting && <Loader2 className="h-4 w-4 animate-spin" />}
              {submitting ? "Creating monitor" : "Create monitor"}
            </button>
          </div>
        </form>
      </div>
    </div>
  );
}

import type { Alert } from "@/lib/api";
import { fullDate, timeAgo } from "@/lib/format";
import Skeleton from "./Skeleton";

export default function AlertItem({
  alert,
  showUrl = false,
}: {
  alert: Alert;
  showUrl?: boolean;
}) {
  return (

        )}
      </div>
    </li>
  );
}

  const [confirming, setConfirming] = useState(false);

  useEffect(() => {

    const t = setTimeout(() => setConfirming(false), 3000);
    return () => clearTimeout(t);

      setChecking(false);

      setConfirming(false);

export const API_BASE = (
  process.env.NEXT_PUBLIC_API_URL || "http://localhost:3000"
).replace(/\/+$/, "");

const API_KEY = process.env.NEXT_PUBLIC_API_KEY || "";

export class ApiError extends Error {
  status: number;

  constructor(message: string, status: number) {
    super(message);
    this.name = "ApiError";
    this.status = status;
  }
}

async function request<T>(
  path: string,
  init: { method?: string; body?: unknown } = {}
): Promise<T> {
  const headers: Record<string, string> = {
    Accept: "application/json",
    "x-api-key": API_KEY,
  };
  // Only send a JSON content type when there is a body; some servers reject
  // empty bodies that claim to be JSON (e.g. on DELETE).
  if (init.body !== undefined) headers["Content-Type"] = "application/json";

  let res: Response;
  try {
    res = await fetch(`${API_BASE}${path}`, {
      method: init.method ?? "GET",
      headers,
      body: init.body !== undefined ? JSON.stringify(init.body) : undefined,
      cache: "no-store",
    });
  } catch {
    throw new ApiError(
      `Can't reach the API at ${API_BASE}. Check that the server is running and that CORS allows this origin and the x-api-key header.`,
      0
    );
  }

  const text = await res.text();
  let data: unknown = undefined;
  if (text) {
    try {
      data = JSON.parse(text);
    } catch {
      data = text;
    }
  }

  if (!res.ok) {
    let message =
      res.status === 401 || res.status === 403
        ? "The API rejected the request. Check NEXT_PUBLIC_API_KEY."
        : `Request failed with status ${res.status}.`;

    if (data && typeof data === "object") {
      const body = data as { message?: unknown; error?: unknown };
      const m = body.message ?? body.error;
      if (typeof m === "string" && m) message = m;
      else if (Array.isArray(m) && m.length) message = m.join(", ");
    }
    throw new ApiError(message, res.status);
  }

  return data as T;
}

===== lib\format.ts =====
import { format, formatDistanceToNow } from "date-fns";

function toDate(value?: string | null): Date | null {
  if (!value) return null;
  const d = new Date(value);
  return Number.isNaN(d.getTime()) ? null : d;
}

export function timeAgo(value?: string | null, fallback = "Never"): string {
  const d = toDate(value);
  return d ? formatDistanceToNow(d, { addSuffix: true }) : fallback;
}

export function fullDate(value?: string | null): string {
  const d = toDate(value);
  return d ? format(d, "PPpp") : "";
}

export function timestamp(value?: string | null): number {
  return toDate(value)?.getTime() ?? 0;
}

export function formatInterval(mins: number): string {
  if (mins % 1440 === 0) {
    const days = mins / 1440;
    return days === 1 ? "daily" : `every ${days} days`;
  }
  if (mins % 60 === 0) {
    const hours = mins / 60;
    return hours === 1 ? "hourly" : `every ${hours} hours`;
  }
  return `every ${mins} min`;
}
